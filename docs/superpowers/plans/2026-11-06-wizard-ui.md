# Assistant de transcription et thème DSFR — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Remplacer l'interface mono-page par un assistant à six écrans (Clé conditionnelle, prompt, réglages, progression, fin), ajouter une page Options, et appliquer un thème Qt léger inspiré du DSFR.

**Architecture:** Un `MainWindow` shell/contrôleur possède un `WizardState`, un `QStackedWidget` de pages, un stepper, un worker `QThread` et une sonde de durée asynchrone. Les pages sont des widgets autonomes qui chargent/commitent l'état et émettent des signaux ; le cœur `TranscribeService`, le CLI, les writers et le packaging restent inchangés. Le thème est un module QSS/palette original appliqué sur `Fusion`, sans asset ni police officielle DSFR.

**Tech Stack:** Python ≥ 3.13, PySide6 6.7–6.9, pytest, pytest-qt, uv.

**Spec:** `docs/superpowers/specs/2026-11-06-wizard-ui-design.md`

## Global Constraints

- Interface en français ; identifiants, modules, classes et signaux en anglais.
- Python ≥ 3.13 ; `from __future__ import annotations` en tête de chaque module.
- Aucune nouvelle dépendance, aucun asset, aucune police embarquée, aucun fichier CSS externe.
- Le DSFR est une inspiration non officielle : pas de code/composant/logo/police DSFR, pas de revendication de conformité.
- Le cœur `core/`, le CLI, `cli.py`, les writers et `packaging/` ne changent pas.
- Une seule transcription à la fois ; aucune opération > 100 ms sur le thread UI.
- La clé OpenRouter n'apparaît jamais dans `QSettings`, les logs ou les détails d'erreur.
- Les tests ne font aucun appel réseau, n'utilisent pas le keyring réel et n'appellent pas ffprobe réel.
- Les sorties SRT/TXT et l'écriture atomique restent inchangées.
- Version cible **0.3.0** dans `pyproject.toml` et `src/ecotranscribe/__init__.py`.
- `uv run pytest` complet avant livraison ; `QT_QPA_PLATFORM=offscreen` si nécessaire.
- Aucun artefact généré, secret ou modification manuelle de `uv.lock` dans le diff.
- Créer une branche de travail (`feat/wizard-ui`) ou un worktree via `using-git-worktrees` avant la première tâche.

# src/ecotranscribe/gui/duration.py
"""Mesure asynchrone de la durée d'un média via ffprobe."""

from __future__ import annotations

import logging
from collections.abc import Callable
from pathlib import Path

from PySide6.QtCore import QObject, QThread, Signal, Slot

from ..core.media import get_media_duration
from ..core.models import CancelToken

logger = logging.getLogger("ecotranscribe")

ProbeFn = Callable[[Path, CancelToken], float | None]


class _ProbeWorker(QObject):
    done = Signal(object, object)
    failed = Signal(object, str)

    def __init__(self, path: Path, probe: ProbeFn, cancel: CancelToken) -> None:
        super().__init__()
        self._path = path
        self._probe = probe
        self._cancel = cancel

    @Slot()
    def run(self) -> None:
        try:
            value = self._probe(self._path, self._cancel)
        except Exception as exc:
            logger.warning("Durée indisponible pour %s : %s", self._path, exc)
            self.failed.emit(self._path, str(exc))
        else:
            self.done.emit(self._path, value)


class DurationProbe(QObject):
    ready = Signal(object, object)
    failed = Signal(object, str)

    def __init__(
        self,
        probe: ProbeFn | None = None,
        parent: QObject | None = None,
    ) -> None:
        super().__init__(parent)
        self._probe = probe or _default_probe
        self._current: Path | None = None
        self._cancel: CancelToken | None = None
        self._threads: set[tuple[QThread, _ProbeWorker]] = set()

    def start(self, path: Path) -> None:
        self.cancel()
        path = Path(path)
        self._current = path
        cancel = CancelToken()
        thread = QThread(self)
        worker = _ProbeWorker(path, self._probe, cancel)
        worker.moveToThread(thread)
        thread.started.connect(worker.run)
        worker.done.connect(self._on_done)
        worker.failed.connect(self._on_failed)
        worker.done.connect(thread.quit)
        worker.failed.connect(thread.quit)
        thread.finished.connect(worker.deleteLater)
        thread.finished.connect(thread.deleteLater)
        job = (thread, worker)
        self._threads.add(job)
        thread.finished.connect(lambda: self._threads.discard(job))
        self._cancel = cancel
        thread.start()

    def cancel(self) -> None:
        if self._cancel is not None:
            self._cancel.cancel()
        self._cancel = None
        self._current = None

    def shutdown(self) -> None:
        self.cancel()
        for thread, _worker in list(self._threads):
            thread.quit()
            thread.wait(1000)

    def _on_done(self, path: object, value: object) -> None:
        path = Path(path)
        if self._current != path:
            return
        self._current = None
        self._cancel = None
        self.ready.emit(path, None if value is None else float(value))

    def _on_failed(self, path: object, message: str) -> None:
        path = Path(path)
        if self._current != path:
            return
        self._current = None
        self._cancel = None
        self.failed.emit(path, message)


def _default_probe(path: Path, cancel: CancelToken) -> float | None:
    return get_media_duration(path, cancel=cancel)
```

from __future__ import annotations

import pytest
from PySide6.QtCore import Qt
from PySide6.QtGui import QPalette
from PySide6.QtWidgets import QApplication

from ecotranscribe.gui.theme import DARK, LIGHT, ThemeTokens, apply_theme


def _luminance(value: str) -> float:
    raw = value.lstrip("#")
    channels = [int(raw[i:i + 2], 16) / 255 for i in (0, 2, 4)]
    linear = [
        c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4
        for c in channels
    ]
    return 0.2126 * linear[0] + 0.7152 * linear[1] + 0.0722 * linear[2]


def _contrast(a: str, b: str) -> float:
    high, low = sorted((_luminance(a), _luminance(b)), reverse=True)
    return (high + 0.05) / (low + 0.05)


@pytest.mark.parametrize("tokens", [LIGHT, DARK])
def test_text_contrast_is_accessible(tokens: ThemeTokens) -> None:
    assert _contrast(tokens.text, tokens.background) >= 4.5

def _palette(tokens: ThemeTokens) -> QPalette:
    palette = QPalette()
    roles = {
        QPalette.ColorRole.Window: tokens.background,
        QPalette.ColorRole.WindowText: tokens.text,
        QPalette.ColorRole.Base: tokens.surface,
        QPalette.ColorRole.AlternateBase: tokens.surface_alt,
        QPalette.ColorRole.Text: tokens.text,
        QPalette.ColorRole.Button: tokens.surface,
        QPalette.ColorRole.ButtonText: tokens.text,
        QPalette.ColorRole.Highlight: tokens.primary,
        QPalette.ColorRole.HighlightedText: tokens.on_primary,
        QPalette.ColorRole.Link: tokens.primary,
        QPalette.ColorRole.PlaceholderText: tokens.text_muted,
        QPalette.ColorRole.ToolTipBase: tokens.surface_contrast,
        QPalette.ColorRole.ToolTipText: tokens.text,
    }
    for role, value in roles.items():
        palette.setColor(role, QColor(value))
    for role in (
        QPalette.ColorRole.WindowText,
        QPalette.ColorRole.Text,
        QPalette.ColorRole.ButtonText,
    ):
        palette.setColor(
            QPalette.ColorGroup.Disabled, role, QColor(tokens.text_disabled)
        )
    return palette


def apply_theme(
    app: QApplication,
    scheme: Qt.ColorScheme | None = None,
) -> None:
    app.setStyle("Fusion")
    if scheme is None:
        scheme = QGuiApplication.styleHints().colorScheme()
    tokens = DARK if scheme == Qt.ColorScheme.Dark else LIGHT
    app.setPalette(_palette(tokens))
    app.setStyleSheet(stylesheet(tokens))


class ThemeManager(QObject):
    def __init__(self, app: QApplication) -> None:
        super().__init__(app)
        self._app = app
        QGuiApplication.styleHints().colorSchemeChanged.connect(self.apply)

    def apply(self, *_args: object) -> None:
        apply_theme(self._app)


def _repolish(widget: QWidget) -> None:
    style = widget.style()
    style.unpolish(widget)
    style.polish(widget)
    widget.update()


def repolish(widget: QWidget) -> None:
    _repolish(widget)


def set_variant(widget: QWidget, value: str) -> None:
    widget.setProperty("variant", value)
    _repolish(widget)


def set_role(widget: QWidget, value: str) -> None:
    widget.setProperty("role", value)
    _repolish(widget)


def set_severity(widget: QWidget, value: str) -> None:
    widget.setProperty("severity", value)
    _repolish(widget)


def set_state(widget: QWidget, value: str) -> None:
    widget.setProperty("state", value)
    _repolish(widget)

class WizardPage(QWidget):
    def __init__(self, title: str, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        root = QVBoxLayout(self)
        root.setContentsMargins(32, 24, 32, 24)
        root.setSpacing(20)
        self.title_label = QLabel(title)
        self.title_label.setProperty("role", "pageTitle")
        root.addWidget(self.title_label)

        scroll = QScrollArea()
        scroll.setWidgetResizable(True)
        scroll.setFrameShape(QFrame.Shape.NoFrame)
        self.content = QWidget()
        self.content_layout = QVBoxLayout(self.content)
        self.content_layout.setContentsMargins(0, 0, 0, 0)
        self.content_layout.setSpacing(16)
        scroll.setWidget(self.content)
        root.addWidget(scroll, 1)

        self.action_bar = QWidget()
        actions = QHBoxLayout(self.action_bar)
        actions.setContentsMargins(0, 0, 0, 0)
        self._left = QHBoxLayout()
        self._right = QHBoxLayout()
        actions.addLayout(self._left)
        actions.addStretch(1)
        actions.addLayout(self._right)
        root.addWidget(self.action_bar)

    def set_title(self, text: str) -> None:
        self.title_label.setText(text)

    def add_secondary(self, button: QPushButton) -> None:
        self._left.addWidget(button)

    def add_primary(self, button: QPushButton) -> None:
        self._right.addWidget(button)

    def focus_primary(self) -> None:
        for layout in (self._right, self._left):
            for index in range(layout.count()):
                widget = layout.itemAt(index).widget()
                if isinstance(widget, QPushButton) and widget.isEnabled():
                    widget.setFocus()
                    return

        layout.addWidget(self.description_label)
        layout.addWidget(self.path_label)
        layout.addLayout(buttons)

        for label in (self.title_label, self.description_label, self.path_label):
            label.setAttribute(Qt.WidgetAttribute.WA_TransparentForMouseEvents)
        self.browse_button.clicked.connect(self.browse_requested)
        self.continue_button.clicked.connect(self.continue_requested)
        self.change_button.clicked.connect(self.change_requested)

        self.continue_button.setVisible(has_file)

        if event.button() == Qt.MouseButton.LeftButton:
            self.clicked.emit()
        super().mouseReleaseEvent(event)

    def keyPressEvent(self, event: QKeyEvent) -> None:
        if event.key() in (Qt.Key.Key_Return, Qt.Key.Key_Enter, Qt.Key.Key_Space):
            self.clicked.emit()

# src/ecotranscribe/gui/widgets/key_form.py
"""Formulaire de clé partagé par l'écran Clé et la page Options."""

from __future__ import annotations

from PySide6.QtCore import QUrl, Signal
from PySide6.QtGui import QDesktopServices
from PySide6.QtWidgets import (
    QCheckBox,
    QLabel,
    QLineEdit,
    QPushButton,
    QVBoxLayout,
    QWidget,
)

OPENROUTER_KEYS_URL = "https://openrouter.ai/keys"


class KeyForm(QWidget):
    changed = Signal()

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self._dirty = False
        self._loading = False
        layout = QVBoxLayout(self)
        layout.setContentsMargins(0, 0, 0, 0)
        layout.setSpacing(8)

        self.status_label = QLabel("")
        self.status_label.setProperty("role", "muted")
        self.status_label.setWordWrap(True)
        label = QLabel("Nouvelle clé OpenRouter")
        label.setProperty("role", "label")
        self.key_edit = QLineEdit()
        self.key_edit.setEchoMode(QLineEdit.EchoMode.Password)
        self.key_edit.setPlaceholderText("Collez votre clé OpenRouter")
        self.remember = QCheckBox("Mémoriser dans le trousseau")
        self.remember.setChecked(True)
        self.link = QPushButton("Où trouver ma clé ?")
        self.link.setProperty("variant", "tertiary")
        self.link.clicked.connect(self._open_keys_page)
        self.note = QLabel(
            "La clé n'est jamais écrite dans les réglages ni dans le journal."
        )
        self.note.setProperty("role", "mention")
        self.note.setWordWrap(True)
        self.error_label = QLabel("")
        self.error_label.setProperty("role", "error")
        self.error_label.setWordWrap(True)
        self.error_label.setVisible(False)

        layout.addWidget(self.status_label)
        layout.addWidget(label)
        layout.addWidget(self.key_edit)
        layout.addWidget(self.remember)
        layout.addWidget(self.link)
        layout.addWidget(self.note)
        layout.addWidget(self.error_label)

        self.key_edit.textChanged.connect(self._on_changed)
        self.remember.toggled.connect(self._on_changed)

    def key(self) -> str:
        return self.key_edit.text().strip()

    def remember_key(self) -> bool:
        return self.remember.isEnabled() and self.remember.isChecked()

    def set_status(self, text: str) -> None:
        self.status_label.setText(text)

    def set_keyring_available(self, available: bool) -> None:
        self._loading = True
        self.remember.setEnabled(available)
        if not available:
            self.remember.setChecked(False)
            self.remember.setToolTip(
                "Le trousseau système est indisponible ; la clé restera "
                "valable pour cette session."
            )
        else:
            self.remember.setToolTip("")
        self._loading = False

    def set_error(self, text: str) -> None:
        self.error_label.setText(text)
        self.error_label.setVisible(bool(text))

    def clear_field(self) -> None:
        self._loading = True
        self.key_edit.clear()
        self.error_label.clear()
        self.error_label.setVisible(False)
        self._loading = False

    def mark_clean(self) -> None:
        self._dirty = False

    def is_dirty(self) -> bool:
        return self._dirty

    def _open_keys_page(self) -> None:
        QDesktopServices.openUrl(QUrl(OPENROUTER_KEYS_URL))

    def _on_changed(self, *_args: object) -> None:
        if not self._loading:
            self._dirty = True
            self.changed.emit()
```

# src/ecotranscribe/gui/widgets/stepper.py
"""Indicateur d'étapes non cliquable, inspiré du composant DSFR."""

from __future__ import annotations

from PySide6.QtCore import Qt
from PySide6.QtWidgets import QFrame, QHBoxLayout, QLabel, QVBoxLayout, QWidget

from ..theme import repolish


class Stepper(QWidget):
    STEPS = ("Fichier", "Prompt", "Réglages", "Transcription")

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self.current = 0
        self._numbers: list[QLabel] = []
        self._titles: list[QLabel] = []
        self._lines: list[QFrame] = []
        layout = QHBoxLayout(self)
        layout.setContentsMargins(0, 0, 0, 0)
        layout.setSpacing(8)
        for index, title in enumerate(self.STEPS):
            column = QVBoxLayout()
            number = QLabel(str(index + 1))
            number.setAlignment(Qt.AlignmentFlag.AlignCenter)
            title_label = QLabel(title)
            title_label.setProperty("role", "mention")
            self._numbers.append(number)
            self._titles.append(title_label)
            column.addWidget(number, 0, Qt.AlignmentFlag.AlignHCenter)
            column.addWidget(title_label, 0, Qt.AlignmentFlag.AlignHCenter)
            layout.addLayout(column)
            if index < len(self.STEPS) - 1:
                line = QFrame()
                line.setFixedHeight(2)
                self._lines.append(line)
                layout.addWidget(line, 1)
        self.set_current(0)

    def set_current(self, index: int) -> None:
        self.current = max(0, min(index, len(self.STEPS) - 1))
        for position, number in enumerate(self._numbers):
            if position < self.current:
                state = "done"
            elif position == self.current:
                state = "current"
            else:
                state = "future"
            number.setProperty("stepNumber", state)
            self._titles[position].setProperty("stepTitle", state)
            repolish(number)
            repolish(self._titles[position])
        for position, line in enumerate(self._lines):
            line.setProperty("stepLine", "done" if position < self.current else "future")
            repolish(line)
        self.setAccessibleName(
            f"Étape {self.current + 1} sur {len(self.STEPS)} : "
            f"{self.STEPS[self.current]}"
        )

"""Porte conditionnelle de la clé OpenRouter."""

from __future__ import annotations

from enum import Enum

from PySide6.QtCore import Signal
from PySide6.QtWidgets import QLabel, QPushButton, QWidget

from ..widgets.key_form import KeyForm
from ..widgets.layout import WizardPage

_INITIAL_COPY = (
    "EcoTranscribe utilise OpenRouter pour transcrire. "
    "Saisissez votre clé API pour continuer."
)
_CORRECTION_COPY = (
    "La clé a été refusée. Saisissez une nouvelle clé, puis relancez "
    "la transcription depuis les réglages."
)


class KeyMode(str, Enum):
    INITIAL = "initial"
    CORRECTION = "correction"


class KeyPage(WizardPage):
    submitted = Signal(str, bool)
    back_requested = Signal()

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__("Clé OpenRouter", parent)
        self.intro = QLabel(_INITIAL_COPY)
        self.intro.setWordWrap(True)
        self.key_form = KeyForm()
        self.content_layout.addWidget(self.intro)
        self.content_layout.addWidget(self.key_form)
        self.content_layout.addStretch(1)
        self.next_button = QPushButton("Suivant")
        self.next_button.setProperty("variant", "primary")
        self.back_button = QPushButton("Retour")
        self.back_button.setProperty("variant", "secondary")
        self.add_secondary(self.back_button)
        self.add_primary(self.next_button)
        self.next_button.clicked.connect(self._submit)
        self.back_button.clicked.connect(self.back_requested)
        self.key_form.set_status("Aucune clé active")

    def _submit(self) -> None:
        key = self.key_form.key()
        if not key:
            self.key_form.set_error("Saisissez une clé OpenRouter.")

    def load(self, state: WizardState) -> None:

        self.message.setVisible(False)
        self.use_source.setVisible(False)
        for model in OPENROUTER_MODELS:

        self.txt.setChecked(OutputFormat.TXT in state.formats)
        self._user_set = state.output_dir_user_set
        if state.output_dir_user_set and state.output_dir is not None:
            self.output_edit.setText(str(state.output_dir))
        elif state.input_path is not None:
            self.output_edit.setText(str(state.input_path.parent))
        elif state.output_dir is not None:
            self.output_edit.setText(str(state.output_dir))
        else:
            self.output_edit.clear()

    def commit(self, state: WizardState) -> bool:
        self.clear_errors()
        self.message.setVisible(False)
        valid = True
        model = self._current_model()
        if not model:
            self._set_error(self.model_error, "Choisissez un modèle.")
            valid = False
        formats = self._selected_formats()
        if not formats:
            self._set_error(
                self.formats_error,
                "Choisissez au moins un format de sortie.",
            )
            valid = False
        output_text = self.output_edit.text().strip()
        if output_text:
            output_dir: Path | None = Path(output_text).expanduser()
        elif state.input_path is not None:
            output_dir = state.input_path.parent
        else:
            output_dir = None
        if output_dir is None:
            self._set_error(
                self.output_error,
                "Choisissez un dossier de sortie.",
            )
            self.use_source.setVisible(False)
            valid = False
        elif not output_dir.exists():
            self._set_error(
                self.output_error,
                "Le dossier de sortie n'existe pas.",
            )
            self.use_source.setVisible(state.input_path is not None)
            valid = False
        elif not output_dir.is_dir():
            self._set_error(
                self.output_error,
                "Le dossier de sortie n'est pas un dossier.",
            )
            self.use_source.setVisible(False)
            valid = False
        elif not os.access(output_dir, os.W_OK | os.X_OK):
            self._set_error(
                self.output_error,
                "Le dossier de sortie n'est pas inscriptible.",
            )
            self.use_source.setVisible(False)
            valid = False
        if not valid:
            return False
        self.capture(state)
        return True

    def capture(self, state: WizardState) -> None:
        state.model = self._current_model() or state.model
        state.language = self.language_combo.currentData()
        formats = self._selected_formats()
        if formats:
            state.formats = formats
        output_text = self.output_edit.text().strip()
        if output_text:
            state.output_dir = Path(output_text).expanduser()
            state.output_dir_user_set = self._user_set
        elif state.input_path is not None:
            state.output_dir = state.input_path.parent
            state.output_dir_user_set = False
        else:
            state.output_dir = None
            state.output_dir_user_set = False

    def set_price(self, model: str, price: float) -> None:
        index = self._model_index(model)
        if index < 0:
            return
        old_text = self.model_combo.itemText(index)
        current_index = self.model_combo.currentIndex()
        current_text = self.model_combo.currentText()
        self.model_combo.setItemText(
            index, f"{model} ({format_hourly_price(price)})"
        )
        if current_index == index and current_text == old_text:
            self.model_combo.setEditText(current_text)

    def set_output_dir(self, path: Path, *, user_set: bool = True) -> None:
        self._user_set = user_set
        self.output_edit.setText(str(path))
        self._set_error(self.output_error, "")
        self.use_source.setVisible(False)

    def show_message(self, text: str) -> None:
        self.message.setText(text)
        self.message.setVisible(bool(text))

    def clear_errors(self) -> None:
        for label in (self.model_error, self.formats_error, self.output_error):
            label.clear()
            label.setVisible(False)

    def focus_primary(self) -> None:
        self.model_combo.setFocus()

    def add_model(self, model: str) -> None:
        self._add_model(model)

    def _add_model(self, model: str) -> None:
        model = (model or "").strip()
        if model and self._model_index(model) < 0:
            self.model_combo.addItem(model, model)

    def _model_index(self, model: str) -> int:
        wanted = (model or "").strip()
        for index in range(self.model_combo.count()):
            if self.model_combo.itemData(index) == wanted:
                return index
        return -1

    def _current_model(self) -> str:
        index = self.model_combo.currentIndex()
        if index >= 0:
            text = self.model_combo.currentText().strip()
            if text == self.model_combo.itemText(index).strip():
                data = self.model_combo.itemData(index)
                if isinstance(data, str) and data:
                    return data
        return self.model_combo.currentText().strip()

    def _selected_formats(self) -> frozenset[OutputFormat]:
        formats = set()
        if self.srt.isChecked():
            formats.add(OutputFormat.SRT)
        if self.txt.isChecked():
            formats.add(OutputFormat.TXT)
        return frozenset(formats)

    def _on_output_edited(self) -> None:
        self._user_set = True
        self.use_source.setVisible(False)
        self._set_error(self.output_error, "")

    def _clear_format_error(self) -> None:
        self._set_error(self.formats_error, "")

    @staticmethod
    def _set_error(label: QLabel, message: str) -> None:
        label.setText(message)
        label.setVisible(bool(message))

# src/ecotranscribe/gui/pages/progress.py
"""Écran de progression, journal, annulation et erreurs inline."""

from __future__ import annotations

import logging
import time

from PySide6.QtCore import QTimer, Signal
from PySide6.QtWidgets import (
    QLabel,
    QPlainTextEdit,
    QProgressBar,
    QPushButton,
    QWidget,

from ...core.errors import TranscribeError
from ...core.formatting import format_seconds
from ...core.models import ProgressUpdate
from ..widgets.alerts import Alert
from ..widgets.layout import WizardPage

logger = logging.getLogger("ecotranscribe")

_PHASE_LABELS = {
    "validating": "Validation",
    "extracting": "Extraction audio",
    "detecting_silences": "Détection des silences",
    "transcribing": "Transcription",
    "writing": "Écriture des fichiers",
    "done": "Terminé",
}


class ProgressPage(WizardPage):
    cancel_requested = Signal()
    back_to_settings_requested = Signal()
    home_requested = Signal()
    open_log_requested = Signal()
    open_folder_requested = Signal()
    edit_key_requested = Signal()
    change_file_requested = Signal()

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__("Transcription", parent)
        self.alert: Alert | None = None
        self._busy = False
        self._writing = False
        self._cancelling = False
        self._action_buttons: list[QPushButton] = []
        self._last_progress: ProgressUpdate | None = None
        self._task_started = 0.0

        self.file_label = QLabel("")
        self.file_label.setProperty("role", "sectionTitle")
        self.phase_label = QLabel("")
        self.phase_label.setWordWrap(True)
        self.progress = QProgressBar()
        self.progress.setProperty("role", "transcription")
        self.progress.setTextVisible(False)
        self.progress.setRange(0, 100)

        self.log = QPlainTextEdit()
        self.log.setReadOnly(True)
        self.log.setProperty("role", "log")
        self.log.setMaximumBlockCount(5000)
        self.log.setMinimumHeight(180)
        self.log_button = QPushButton("Ouvrir le journal")
        self.log_button.setProperty("variant", "tertiary")

        self.content_layout.addWidget(self.file_label)
        self.content_layout.addWidget(self.phase_label)
        self.content_layout.addWidget(self.progress)
        self.content_layout.addWidget(self.elapsed_label)
        self.content_layout.addWidget(self.log, 1)
        self.content_layout.addWidget(self.log_button)

        self.cancel_button = QPushButton("Annuler la transcription")
        self.cancel_button.setProperty("variant", "danger")
        self.add_secondary(self.cancel_button)
        self.cancel_button.clicked.connect(self.cancel_requested)
        self.log_button.clicked.connect(self.open_log_requested)

        self._timer = QTimer(self)
        self._timer.setInterval(1000)
        self._timer.timeout.connect(self._update_elapsed)

    def start_task(self, file_name: str) -> None:
        self.reset()
        self.file_label.setText(file_name)
        self.phase_label.setText("Validation…")
        self.progress.setRange(0, 100)
        self.progress.setValue(0)
        self.cancel_button.setEnabled(True)
        self.cancel_button.setVisible(True)
        self._task_started = time.monotonic()
        self._timer.start()
        self._busy = True
        self._writing = False
        self._cancelling = False

    def update_progress(self, update: ProgressUpdate) -> None:
        self._last_progress = update
        if update.phase == "writing":
            self.show_writing()
            return
        if update.phase == "done":
            self.progress.setRange(0, 1000)
            self.progress.setValue(1000)
            self.phase_label.setText("Terminé")
            self._update_elapsed()
            self._stop_timer()
            return
        label = _PHASE_LABELS.get(update.phase, update.phase)
        detail = update.detail.strip()
        if update.total is not None and update.total > 0:
            ratio = min(max(update.current, 0.0), update.total) / update.total
            percent = round(ratio * 1000)
            self.progress.setRange(0, 1000)
            self.progress.setValue(percent)
            audio = (
                f"{format_seconds(update.current)} / "
                f"{format_seconds(update.total)}"
            )
            parts = [label, audio, f"{percent // 10} %"]
            if detail and detail != label:
                parts.append(detail)
            self.phase_label.setText(" — ".join(parts))
        else:
            self.progress.setRange(0, 0)
            self.phase_label.setText(
                f"{label} — {detail}" if detail and detail != label else label
            )
        if self._cancelling:
            self.phase_label.setText("Annulation en cours…")
        self._update_elapsed()

    def append_log(self, text: str) -> None:
        self.log.appendPlainText(text)
        logger.info("%s", text)

    def show_writing(self) -> None:
        self._writing = True
        self._busy = True
        self.progress.setRange(0, 0)
        self.phase_label.setText("Écriture des fichiers…")
        self.cancel_button.setEnabled(False)
        self._update_elapsed()

    def show_cancelling(self) -> None:
        self._cancelling = True
        self.phase_label.setText("Annulation en cours…")
        self.cancel_button.setEnabled(False)

    def show_error(
        self,
        error: TranscribeError,
        *,
        can_edit_key: bool,
        can_change_file: bool,
        can_open_folder: bool,
    ) -> None:
        self._stop_timer()
        self._busy = True
        self._writing = False
        self._cancelling = False
        self.cancel_button.setVisible(False)
        self.progress.setRange(0, 100)
        self.progress.setValue(0)
        self.phase_label.clear()
        self.elapsed_label.clear()
        text = error.user_message
        if error.hint:
            text += f"\n\n{error.hint}"
        alert = Alert("error", "La transcription a échoué.", text)
        alert.set_detail(error.detail)
        alert.add_action(
            self._action("Accueil", "tertiary", self.home_requested)
        )
        if can_change_file:
            alert.add_action(
                self._action(
                    "Changer de fichier",
                    "secondary",
                    self.change_file_requested,
                )
            )
        if can_edit_key:
            alert.add_action(
                self._action(
                    "Modifier la clé", "secondary", self.edit_key_requested
                )
            )
        if can_open_folder:
            alert.add_action(
                self._action(
                    "Ouvrir le dossier", "secondary", self.open_folder_requested
                )
            )
        alert.add_action(
            self._action(
                "Retour aux réglages",
                "primary",
                self.back_to_settings_requested,
            )
        )
        self._install_alert(alert)

    def show_cancelled(self) -> None:
        self._stop_timer()
        self._busy = True
        self._writing = False
        self._cancelling = False
        self.cancel_button.setVisible(False)
        self.progress.setRange(0, 100)
        self.progress.setValue(0)
        self.phase_label.clear()
        self.elapsed_label.clear()
        alert = Alert(
            "info",
            "Transcription annulée.",
            "Aucun fichier n'a été écrit.",
        )
        alert.add_action(
            self._action("Accueil", "tertiary", self.home_requested)
        )
        alert.add_action(
            self._action(
                "Retour aux réglages",
                "primary",
                self.back_to_settings_requested,
            )
        )
        self._install_alert(alert)

    def set_busy(self, busy: bool) -> None:
        self._busy = busy
        for button in self._action_buttons:
            button.setEnabled(not busy)
        if busy:
            self.cancel_button.setEnabled(
                not self._writing and not self._cancelling
            )
        else:
            self.cancel_button.setEnabled(False)
            if self.alert is not None and self._action_buttons:
                self._action_buttons[-1].setFocus()

    def reset(self) -> None:
        self._stop_timer()
        self._busy = False
        self._writing = False
        self._cancelling = False
        self._last_progress = None
        self.file_label.clear()
        self.phase_label.clear()
        self.elapsed_label.clear()
        self.log.clear()
        self.progress.setRange(0, 100)
        self.progress.setValue(0)
        self.cancel_button.setVisible(True)
        self.cancel_button.setEnabled(False)
        self._clear_alert()

    def focus_primary(self) -> None:
        self.cancel_button.setFocus()

    def _action(self, text: str, variant: str, signal) -> QPushButton:
        button = QPushButton(text)
        button.setProperty("variant", variant)
        button.clicked.connect(signal)
        self._action_buttons.append(button)
        return button

    def _install_alert(self, alert: Alert) -> None:
        self._clear_alert()
        self.content_layout.insertWidget(1, alert)
        self.alert = alert
        self._action_buttons = list(alert.findChildren(QPushButton))

    def _clear_alert(self) -> None:
        if self.alert is not None:
            self.content_layout.removeWidget(self.alert)
            self.alert.deleteLater()
            self.alert = None
        self._action_buttons = []

    def _stop_timer(self) -> None:
        self._timer.stop()
        self._task_started = 0.0

    def _update_elapsed(self) -> None:
        if self._task_started <= 0:
            return
        elapsed = max(0.0, time.monotonic() - self._task_started)
        text = f"Temps écoulé : {format_seconds(elapsed)}"
        progress = self._last_progress
        if (
            progress is not None
            and progress.total is not None
            and progress.total > 0
            and progress.current > 0
        ):
            eta = elapsed * (progress.total - progress.current) / progress.current
            text += f" — ETA : {format_seconds(max(0.0, eta))}"
        self.elapsed_label.setText(text)

# src/ecotranscribe/gui/worker.py
"""Worker QObject exécuté dans un QThread."""

from __future__ import annotations

from PySide6.QtCore import QObject, Signal, Slot

from ..core.errors import CancelledError, TranscribeError
from ..core.models import CancelToken, TranscribeRequest
from ..core.service import TranscribeService


class TranscribeWorker(QObject):
    progress = Signal(object)
    log = Signal(str)
    finished = Signal(object)
    failed = Signal(object)
    cancelled = Signal()

    def __init__(
        self,
        service: TranscribeService,
        request: TranscribeRequest,
        cancel: CancelToken,
        parent: QObject | None = None,
    ) -> None:
        super().__init__(parent)
        self._service = service
        self._request = request
        self._cancel = cancel

    @Slot()
    def run(self) -> None:
        try:
            result = self._service.transcribe(
                self._request,
                on_progress=self.progress.emit,
                on_log=self._on_log,
                cancel=self._cancel,
            )
        except CancelledError:
            self.cancelled.emit()
        except TranscribeError as exc:
            self.failed.emit(exc)
        except Exception as exc:  # garde-fou UI
            self.failed.emit(
                TranscribeError("La transcription a échoué.", detail=str(exc))
            )
        else:
            self.finished.emit(result)

    def _on_log(self, level: str, message: str) -> None:
        if level in ("warning", "error"):
            message = f"{level.upper()} : {message}"
        self.log.emit(message)
```

"""Dialogues injectables de confirmation, isolés du worker et du cœur."""

from __future__ import annotations

from collections.abc import Sequence
from pathlib import Path
from typing import Protocol

from PySide6.QtWidgets import QMessageBox, QWidget


class DialogService(Protocol):
    def confirm_overwrite(self, paths: Sequence[Path]) -> bool: ...

    def confirm_close_while_running(self) -> bool: ...

    def confirm_delete_key(self) -> bool: ...

    def confirm_discard(self, message: str) -> bool: ...


class QtDialogService:
    """Implémentation de production, toujours exécutée sur le thread UI."""

    def __init__(self, parent: QWidget | None = None) -> None:
        self._parent = parent

    def confirm_overwrite(self, paths: Sequence[Path]) -> bool:
        if not paths:
            return True
        box = QMessageBox(self._parent)
        box.setIcon(QMessageBox.Icon.Warning)
        box.setWindowTitle("Fichiers existants")
        box.setText("Ces fichiers existent déjà :")
        box.setInformativeText("\n".join(str(path) for path in paths))
        overwrite = box.addButton("Écraser", QMessageBox.ButtonRole.AcceptRole)
        cancel = box.addButton("Annuler", QMessageBox.ButtonRole.RejectRole)
        box.setDefaultButton(cancel)  # type: ignore[arg-type]
        box.exec()
        return box.clickedButton() is overwrite

    def confirm_close_while_running(self) -> bool:
        answer = QMessageBox.question(
            self._parent,
            "Transcription en cours",
            "Une transcription est en cours. L'annuler et quitter ? "
            "L'arrêt peut attendre la fin de la requête en cours.",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No,
            QMessageBox.StandardButton.No,
        )
        return answer == QMessageBox.StandardButton.Yes

    def confirm_delete_key(self) -> bool:
        answer = QMessageBox.question(
            self._parent,
            "Supprimer la clé",
            "Supprimer la clé OpenRouter mémorisée ?",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No,
            QMessageBox.StandardButton.No,
        )
        return answer == QMessageBox.StandardButton.Yes

    def confirm_discard(self, message: str) -> bool:
        answer = QMessageBox.question(
            self._parent,
            "Abandonner les modifications",
            message,
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No,
            QMessageBox.StandardButton.No,
        )
        return answer == QMessageBox.StandardButton.Yes

        key = key.strip()
        if not key:
            self.key_page.set_error("Saisissez une clé OpenRouter.")
            return
        self.options_page.key_form.remember.setChecked(remember)
        if (
            self._state.session_key == key
            and self._state.key_source is KeySource.SESSION
            and not remember
        ):
            self._after_key()
            return
        if remember:
            try:
                self._key_store.set(key)
            except Exception as exc:
                logger.warning("Écriture du trousseau impossible : %s", exc)
                self._keyring_available = False
                self._state.session_key = key
                self._refresh_key_source()
                self.key_page.set_keyring_available(False)
                self.key_page.set_error(
                    "Impossible d'enregistrer la clé dans le trousseau ; "
                    "elle reste valable pour cette session."
                )
                return
            self._keyring_key = key
            self._state.session_key = None
        else:
            self._state.session_key = key
        self._refresh_key_source()
        self._after_key()

        if path is None:

            return
        if (
            not self._state.prompt_edited
            and self._state.prompt_path is None
            and self._state.prompt_text is None
            and self._state.prompt_error is None
        ):
            self._load_prompt(path)
        self.prompt_page.set_file(path)
        if self._state.duration_path == path:
            self.prompt_page.set_duration(
                self._state.duration_s,
                pending=self._state.duration_pending,
            )
        else:
            self.prompt_page.set_duration(None, pending=True)
        self.prompt_page.set_prompt(
            self._state.prompt_text,
            self._state.prompt_path,
            self._state.prompt_error,
            edited=self._state.prompt_edited,
        )
        self._show_page(PageId.PROMPT, push=push)
        if self._state.duration_path != path and not self._state.duration_pending:
            self._probe_duration(path)

    def _load_prompt(self, path: Path) -> None:
        try:
            prompt_path, prompt_text = resolve_prompt(path)
        except InvalidRequestError as exc:
            self._state.prompt_path = None
            self._state.prompt_text = None
            self._state.prompt_error = exc.user_message
        else:
            self._state.prompt_path = prompt_path
            self._state.prompt_text = prompt_text
            self._state.prompt_error = None
        self._state.prompt_edited = False

    def _probe_duration(self, path: Path) -> None:
        if self._state.duration_path == path and (
            self._state.duration_s is not None
            or self._state.duration_pending
        ):
            return
        self._state.duration_path = path
        self._state.duration_s = None
        self._state.duration_pending = True
        if self._current == PageId.PROMPT and self._state.input_path == path:
            self.prompt_page.set_duration(None, pending=True)
        self._duration_probe.start(path)

    def _on_duration_ready(self, path: object, value: object) -> None:
        path = Path(path)
        if self._state.input_path != path or self._state.duration_path != path:
            return
        self._state.duration_pending = False
        self._state.duration_s = None if value is None else float(value)
        if self._current == PageId.PROMPT:
            self.prompt_page.set_duration(self._state.duration_s, pending=False)

    def _on_duration_failed(self, path: object, message: str) -> None:
        path = Path(path)
        if self._state.input_path != path or self._state.duration_path != path:
            return
        self._state.duration_pending = False
        self._state.duration_s = None
        logger.warning("Durée indisponible pour %s : %s", path, message)
        if self._current == PageId.PROMPT:
            self.prompt_page.set_duration(None, pending=False)

    def _on_prompt_next(self) -> None:
        path = self._state.input_path
        if path is None:
            return
        error = self._validate_input(path)
        if error:
            self.prompt_page.set_error(error)
            return
        self._state.prompt_text = self.prompt_page.prompt_text()
        self._state.prompt_edited = self.prompt_page.prompt_edited()
        self._open_settings()

    # ── Réglages et lancement ──────────────────────────────────────────────

    def _open_settings(self) -> None:
        self._settings_loaded = True
        self.settings_page.load(self._state)
        for model in self._settings.recent_models:
            self.settings_page.add_model(model)
        self._attach_pricing()
        self._show_page(PageId.SETTINGS)

    def _browse_output(self) -> None:
        start_dir = (
            self.settings_page.output_edit.text().strip()
            or self._settings.last_input_dir
            or str(Path.home())
        )
        path = QFileDialog.getExistingDirectory(
            self, "Choisir le dossier de sortie", start_dir
        )
        if path:
            self.settings_page.set_output_dir(Path(path))

    def _use_source_dir(self) -> None:
        if self._state.input_path is not None:
            self.settings_page.set_output_dir(
                self._state.input_path.parent, user_set=False
            )

    def _launch(self) -> None:
        if self.is_busy:
            return
        if not self.settings_page.commit(self._state):
            return
        if self._state.input_path is None:
            return
        if not self._active_key:
            self.settings_page.show_message(KEY_MISSING_MESSAGE)
            self._go_key(KeyMode.CORRECTION, return_page=PageId.SETTINGS)
            return
        output_dir = self._state.output_dir or self._state.input_path.parent
        paths = output_paths(self._state.input_path, output_dir, self._state.formats)
        existing = existing_outputs(paths)
        if existing and not self._dialogs.confirm_overwrite(existing):
            self.settings_page.show_message(
                "Transcription annulée : les fichiers existants sont conservés."
            )
            return
        self._settings.add_recent_model(self._state.model)
        self.save_settings()
        request = TranscribeRequest(
            input_path=self._state.input_path,
            provider=Provider.OPENROUTER,
            model=self._state.model,
            language=self._state.language,
            prompt_text=self._state.prompt_text,
            output_dir=output_dir,
            formats=self._state.formats,
            overwrite=bool(existing),
            api_key=self._active_key,
        )
        try:
            self._start_worker(request)
        except Exception as exc:
            logger.warning("Démarrage de la tâche impossible : %s", exc)
            self.settings_page.show_message(
                "Impossible de démarrer la transcription."
            )

    def _start_worker(self, request: TranscribeRequest) -> None:
        self._cancel_token = CancelToken()
        service = self._service_factory(request)
        worker = TranscribeWorker(service, request, self._cancel_token)
        thread = QThread(self)
        worker.moveToThread(thread)
        thread.started.connect(worker.run)
        worker.progress.connect(self._on_progress)
        worker.log.connect(self._on_log)
        worker.finished.connect(self._on_finished)
        worker.failed.connect(self._on_failed)
        worker.cancelled.connect(self._on_cancelled)
        for signal in (worker.finished, worker.failed, worker.cancelled):
            signal.connect(thread.quit)
        thread.finished.connect(self._on_thread_finished)
        thread.finished.connect(thread.deleteLater)
        self._worker = worker
        self._thread = thread
        self._busy = True
        self._writing = False
        self._pending_new = False
        self._state.clear_result()
        self.progress_page.start_task(request.input_path.name)
        self._show_page(PageId.PROGRESS)
        thread.start()

    def _on_progress(self, update: ProgressUpdate) -> None:
        if update.phase == "writing":
            self._writing = True
        self.progress_page.update_progress(update)

    def _on_log(self, text: str) -> None:
        self.progress_page.append_log(text)

    def _on_finished(self, result) -> None:
        self._state.result = result
        self._state.error = None
        self._state.cancelled = False
        self.progress_page.update_progress(
            ProgressUpdate("done", detail="Terminé")
        )
        file_name = self._state.input_path.name if self._state.input_path else ""
        self.done_page.show_result(result, file_name)
        self._show_page(PageId.DONE)
        logger.info("Transcription terminée")

    def _on_failed(self, error: TranscribeError) -> None:
        self._state.error = error
        self._state.result = None
        self._state.cancelled = False
        self._writing = False
        can_edit_key = isinstance(error, EngineUnavailableError)
        can_change_file = (
            isinstance(error, InvalidRequestError)
            and "fichier" in error.user_message.lower()
        )
        can_open_folder = (
            self._state.output_dir is not None
            and self._state.output_dir.is_dir()
        )
        self.progress_page.show_error(
            error,
            can_edit_key=can_edit_key,
            can_change_file=can_change_file,
            can_open_folder=can_open_folder,
        )
        self.progress_page.set_busy(self.is_busy)
        logger.warning("Transcription échouée : %s", error.user_message)

    def _on_cancelled(self) -> None:
        self._state.cancelled = True
        self._state.error = None
        self._state.result = None
        self._writing = False
        self.progress_page.show_cancelled()
        self.progress_page.set_busy(self.is_busy)
        logger.info("Transcription annulée")

    def _on_thread_finished(self) -> None:
        self._thread = None
        self._worker = None
        self._cancel_token = None
        self._busy = False
        self._writing = False
        self.progress_page.set_busy(False)
        self._update_gear()
        if self._close_when_done:
            self._close_when_done = False
            self.close()
            return
        if self._pending_new:
            self._pending_new = False
            self._go_home(reset=True)

    def cancel_transcription(self) -> None:
        if (
            self._thread is None
            or self._cancel_token is None
            or self._writing
        ):
            return
        self._cancel_token.cancel()
        self.progress_page.show_cancelling()
        logger.info("Annulation demandée")

    def _back_to_settings(self) -> None:
        if self._thread is not None:
            return
        self._rewind_to(PageId.SETTINGS)

    def _edit_key_after_error(self) -> None:
        if self._thread is not None:
            return
        self._go_key(KeyMode.CORRECTION, return_page=PageId.SETTINGS)

    def _open_log(self) -> None:
        QDesktopServices.openUrl(QUrl.fromLocalFile(str(log_file_path())))

    def _open_output_folder(self) -> None:
        result = self._state.result
        folder = (
            result.output_paths[0].parent
            if result is not None and result.output_paths
            else self._state.output_dir
        )
        if folder is not None and folder.is_dir():
            QDesktopServices.openUrl(QUrl.fromLocalFile(str(folder)))

    # ── Options ────────────────────────────────────────────────────────────

    def _open_options(self) -> None:
        if self.is_busy or self._current == PageId.OPTIONS:
            return
        if self._current == PageId.PROMPT:
            self._capture_prompt()
        if self._current == PageId.SETTINGS:
            self._capture_settings()
        self.options_page.load(
            self._key_status_text(),
            self._keyring_available,
            build_info(),
            self._can_delete_key(),
        )
        self._show_page(PageId.OPTIONS)

    def _save_options_key(self, key: str, remember: bool) -> None:
        key = key.strip()
        if not key:
            self.options_page.set_error("Saisissez une clé OpenRouter.")
            return
        self.key_page.key_form.remember.setChecked(remember)
        if remember:
            try:
                self._key_store.set(key)
            except Exception as exc:
                logger.warning("Écriture du trousseau impossible : %s", exc)
                self._keyring_available = False
                self._state.session_key = key
                self._refresh_key_source()
                self.options_page.set_keyring_available(False)
                self.options_page.set_error(
                    "Impossible d'enregistrer la clé dans le trousseau ; "
                    "elle reste valable pour cette session."
                )
                return
            self._keyring_key = key
            self._state.session_key = None
            self.options_page.show_saved(
                "Clé enregistrée dans le trousseau."
            )
        else:
            self._state.session_key = key
            self.options_page.show_saved(
                "Clé de session enregistrée pour cette exécution."
            )
        self._refresh_key_source()

    def _delete_options_key(self) -> None:
        if not self._dialogs.confirm_delete_key():
            return
        if self._keyring_key is not None:
            try:
                self._key_store.delete()
            except Exception as exc:
                logger.warning("Suppression du trousseau impossible : %s", exc)
                self.options_page.set_error(
                    "Impossible de supprimer la clé du trousseau."
                )
                return
        self._keyring_key = None
        self._state.session_key = None
        self._refresh_key_source()
        self.options_page.clear_field()
        self.options_page.set_error("")
        if self._state.key_source is KeySource.ENV:
            self.options_page.show_saved(
                "Clé du trousseau supprimée ; la clé d'environnement reste "
                "active."
            )
        else:
            self.options_page.show_saved("Clé supprimée.")

    # ── Réglages, fermeture et raccourcis ──────────────────────────────────

    def save_settings(self) -> None:
        if self._settings_loaded:
            self.settings_page.capture(self._state)
        self._settings.geometry = self.saveGeometry()
        if self._state.input_path is not None:
            self._settings.last_input_dir = str(self._state.input_path.parent)
        self._settings.model = self._state.model
        self._settings.language = self._state.language or ""
        self._settings.formats = {
            fmt.value for fmt in self._state.formats
        }
        self._settings.output_dir = (
            str(self._state.output_dir)
            if self._state.output_dir_user_set and self._state.output_dir
            else ""
        )
        self._settings.remember_api_key = (
            self.key_page.key_form.remember.isChecked()
        )
        self._settings.save()

    def closeEvent(self, event: QCloseEvent) -> None:
        if self.is_busy:
            if not self._dialogs.confirm_close_while_running():
                event.ignore()
                return
            self._close_when_done = True
            self.cancel_transcription()
            event.ignore()
            return
        if self._current == PageId.OPTIONS and self.options_page.is_dirty():
            if not self._dialogs.confirm_discard(
                "La clé saisie n'est pas enregistrée. Quitter ?"
            ):
                event.ignore()
                return
        self.save_settings()
        self._duration_probe.shutdown()
        super().closeEvent(event)

    def dragEnterEvent(self, event: QDragEnterEvent) -> None:
        if event.mimeData().hasUrls():
            event.acceptProposedAction()
        else:
            event.ignore()

    def dropEvent(self, event: QDropEvent) -> None:
        if self.is_busy or self._current not in (
            PageId.HOME,
            PageId.PROMPT,
        ):
            event.ignore()
            return
        for url in event.mimeData().urls():
            if url.isLocalFile():
                path = Path(url.toLocalFile())
                if path.is_file():
                    self.select_file(path, confirm_prompt=True)
                    event.acceptProposedAction()
                    return
        event.ignore()

    def _primary_shortcut(self) -> None:
        if self._current == PageId.HOME:
            self._continue_from_home()
        elif self._current == PageId.KEY:
            self.key_page.next_button.click()
        elif self._current == PageId.PROMPT:
            self.prompt_page.next_button.click()
        elif self._current == PageId.SETTINGS:
            self.settings_page.launch_button.click()

    def _escape(self) -> None:
        if self._current == PageId.PROGRESS:
            if self.is_busy:
                self.cancel_transcription()
            else:
                self._back_to_settings()
        elif self._current == PageId.KEY:
            self._on_key_back()
        elif self._current in (
            PageId.PROMPT,
            PageId.SETTINGS,
            PageId.OPTIONS,
        ):
            self._back()
        elif self._current == PageId.HOME:
            self.close()
        elif self._current == PageId.DONE:
            self._go_home(reset=False)

    def _default_service(
        self, request: TranscribeRequest
    ) -> TranscribeService:
        return TranscribeService({Provider.OPENROUTER: OpenRouterEngine()})
```

```python
# src/ecotranscribe/gui/app.py
"""Point d'entrée `ecotranscribe-desktop` et mode `--diagnose`."""

from __future__ import annotations

import argparse
import json
import sys
from collections.abc import Sequence
from pathlib import Path

from PySide6.QtCore import QSettings
from PySide6.QtWidgets import QApplication

from ecotranscribe.core.config import APP_NAME, APP_SLUG
from ecotranscribe.gui.availability import collect_diagnostic
from ecotranscribe.gui.dialogs import QtDialogService
from ecotranscribe.gui.logging_setup import configure_logging
from ecotranscribe.gui.main_window import MainWindow
from ecotranscribe.gui.pricing import OpenRouterPricingClient
from ecotranscribe.gui.settings import AppSettings, KeyringStore
from ecotranscribe.gui.theme import ThemeManager, apply_theme


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="ecotranscribe-desktop",
        description="EcoTranscribe Desktop.",
    )
    parser.add_argument(
        "--diagnose",
        action="store_true",
        help="Écrit un JSON de diagnostic et quitte sans fenêtre.",
    )
    parser.add_argument(
        "--diagnose-output",
        type=Path,
        default=None,
        help="Écrit aussi le diagnostic dans ce fichier.",
    )
    return parser


def run_diagnostic(output: Path | None) -> int:
    configure_logging()
    diagnostic = collect_diagnostic()
    text = json.dumps(diagnostic, ensure_ascii=False, indent=2)
    if sys.stdout is not None:
        print(text)
    if output is not None:
        output = Path(output)
        output.parent.mkdir(parents=True, exist_ok=True)
        output.write_text(text + "\n", encoding="utf-8")
    ok = bool(diagnostic.get("ffmpeg_path")) and bool(
        diagnostic.get("ffprobe_path")
    )
    return 0 if ok else 1


def main(argv: Sequence[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    if args.diagnose:
        return run_diagnostic(args.diagnose_output)

    app = QApplication.instance()
    if app is None:
        app = QApplication(sys.argv[:1])
    app.setApplicationName(APP_NAME)
    app.setOrganizationName(APP_SLUG)

    configure_logging()
    apply_theme(app)
    theme_manager = ThemeManager(app)
    window = MainWindow(
        settings=AppSettings(QSettings()),
        key_store=KeyringStore(),
        dialogs=QtDialogService(),
        pricing_client=OpenRouterPricingClient(),
    )
    app.aboutToQuit.connect(window.save_settings)
    window.show()
    return app.exec()


if __name__ == "__main__":
    raise SystemExit(main())
```

Le `theme_manager` est parenté à `app` via `QObject`, ce qui le maintient en vie pendant `app.exec()`.
